    struct MyStruct myStructOne;

    struct MyStruct myStructTwo;
    myStructTwo.value = 20;

    struct MyStruct* myStructTwoPointer = &myStructTwo;
    myStructTwoPointer->value = 20;  // sama dengan (*myStructTwoPointer).value = 20;

    printf("myStructOne.value = %d\n", myStructOne.value);
    printf("myStructTwo.value = %d\n", myStructTwo.value);

    printf("myStructTwoPointer->value = %d\n", myStructTwoPointer->value);
    printf("(*myStructTwoPointer).value = %d\n", (*myStructTwoPointer).value);

    return 0;
}

#define createJsLikeArray(...) createJsLikeArrayInner(__VA_ARGS__, NULL)
any* createJsLikeArrayInner(any firstArgument, ...) {
    any* newArray = NULL;
    any newArrayLength = 0;

    va_list args;
    va_start(args, firstArgument);

    for (any current = firstArgument; current != -1; current = va_arg(args, any)) {
        newArrayLength++;
        newArray = realloc(newArray, newArrayLength * sizeof(any));
        newArray[newArrayLength - 1] = current;
    }

    va_end(args);

    // Tambahkan sentinel lagi
    newArrayLength++;
    newArray = realloc(newArray, newArrayLength * sizeof(any));
    newArray[newArrayLength - 1] = -1;

    return newArray;
}
